async def admin_handler(request: web.Request) -> web.StreamResponse:
    file_path = os.path.join(WEBSITE_DIR, "panel-frontend.html")
    if os.path.isfile(file_path):
        resp = web.FileResponse(file_path)
        if not request.cookies.get("lasd_admin_auth"):
            resp.set_cookie("lasd_admin_auth", "1422647977880584324", max_age=86400, secure=True, httponly=True, samesite="Lax")
        return resp

    return web.Response(text="Admin panel frontend not found.", status=404)
